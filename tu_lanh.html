import http.server import json import os import re import socketserver import urllib.parse import webbrowser PORT = 8000 FOLDER_PATH = r"D:\HTML" class MyHandler(http.server.SimpleHTTPRequestHandler): def do_GET(self): # API Tìm kiếm file HTML trị bệnh if self.path.startswith("/api/search"): parsed_url = urllib.parse.urlparse(self.path) params = urllib.parse.parse_qs(parsed_url.query) keyword = params.get("keyword", [""])[0].strip().lower() results = [] if keyword and os.path.exists(FOLDER_PATH): for file_name in os.listdir(FOLDER_PATH): if file_name.endswith(".html") or file_name.endswith( ".htm" ): file_path = os.path.join(FOLDER_PATH, file_name) try: with open(file_path, "r", encoding="utf-8") as f: content = f.read() if keyword in content.lower(): clean_text = re.sub(r"<[^>]*>", "", content) start_idx = clean_text.lower().find(keyword) preview = clean_text[ max(0, start_idx - 20) : min( len(clean_text), start_idx + 80 ) ].strip() results.append( {"file_name": file_name, "preview": preview} ) except: pass self.send_response(200) self.send_header("Content-Type", "application/json") self.end_headers() self.wfile.write(json.dumps(results).encode("utf-8")) return # Giao diện chính tích hợp Tủ lạnh và Tìm kiếm if self.path == "/" or self.path == "/index.html": self.send_response(200) self.send_header("Content-Type", "text/html; charset=utf-8") self.end_headers() html_content = """ Quản Lý Thông Minh - Nhà Nam

🍎 Tủ Lạnh Nhà Nam

🟢 CÒN TRONG TỦ:

Tên mónGiáHết hạnHành động

🔴 CHƯA CÓ / CẦN MUA THÊM: 0

🔍 Bộ Lọc Tìm Kiếm Bệnh (D:\\HTML)

Sẵn sàng...
""" self.wfile.write(html_content.encode("utf-8")) return return http.server.SimpleHTTPRequestHandler.do_GET(self) print(f"Máy chủ Nhà Nam đang chạy tại http://localhost:{PORT}") webbrowser.open(f"http://localhost:{PORT}") with socketserver.TCPServer(("", PORT), MyHandler) as httpd: httpd.serve_forever()
